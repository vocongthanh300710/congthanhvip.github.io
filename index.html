TRIP01 — Hành trình quay về đỉnh 1

Cho đồ thị có hướng N đỉnh, M cung. Mỗi đỉnh có trọng số c[i]. Mỗi cung tốn 1 đơn vị thời gian. Xuất phát đỉnh 1 lúc t=0, quay về đỉnh 1 đúng lúc T. Mỗi lần ghé đỉnh nhận c[i]. Tìm tổng phần thưởng lớn nhất, hoặc -1.

Ý tưởng: Ma trận M[u][v] = c[v] nếu có cung u→v. Khi đó Mk[u][v] = điểm lớn nhất đi u→v qua đúng k cạnh. Đáp án = MT[1][1] + c[1]. Dùng nhân max-plus + luỹ thừa nhanh O(N³ log T).

Code C++

#include<bits/stdc++.h>
#define ll long long
using namespace std;
const int N = 252;
const ll INF = 1e18;
int n, m;
ll t, c[N], a[N][N], b[N][N], d[N][N];
void mul(ll x[N][N], ll y[N][N], ll z[N][N]){
    for(int i = 1; i <= n; i++)
        for(int j = 1; j <= n; j++){
            z[i][j] = -INF;
            for(int k = 1; k <= n; k++)
                if(x[i][k] != -INF && y[k][j] != -INF)
                    z[i][j] = max(z[i][j], x[i][k] + y[k][j]);
        }
}
void init(){
    ios_base::sync_with_stdio(false);
    cin >> n >> m >> t;
    for(int i = 1; i <= n; i++) cin >> c[i];
    for(int i = 1; i <= n; i++)
        for(int j = 1; j <= n; j++) a[i][j] = -INF;
    for(int i = 1; i <= m; i++){
        int u, v; cin >> u >> v;
        a[u][v] = max(a[u][v], c[v]);
    }
}
void solve(){
    for(int i = 1; i <= n; i++)
        for(int j = 1; j <= n; j++) b[i][j] = (i == j ? 0 : -INF);
    while(t > 0){
        if(t & 1){
            mul(b, a, d);
            for(int i = 1; i <= n; i++)
                for(int j = 1; j <= n; j++) b[i][j] = d[i][j];
        }
        mul(a, a, d);
        for(int i = 1; i <= n; i++)
            for(int j = 1; j <= n; j++) a[i][j] = d[i][j];
        t >>= 1;
    }
    if(b[1][1] == -INF) cout << -1;
    else cout << b[1][1] + c[1];
}
int main(){ init(); solve(); }

TRIP02 — Hành trình có sự kiện đặc biệt

Giống TRIP01 nhưng mỗi cung có trọng số thời gian w[i] (1…5), và có k sự kiện đặc biệt (t, x, y): nếu ở đỉnh x đúng lúc t thì cộng thêm y (nếu đang đi trên đường thì không nhận).

Ràng buộcGiá trị
N≤ 50
M≤ 550
K≤ 200
T≤ 10⁹
w[i]1 … 5
y[i]≤ 10⁹
Ý tưởng: DP dp[t][v] = điểm tốt nhất đến v lúc t. Vì T quá lớn, chuyển sang ma trận max-plus kích thước 5n (nhớ 5 lớp thời gian gần nhất). Sự kiện đặc biệt = ma trận riêng cộng thêm y vào các cạnh đi vào x. Tiền xử lý M^(2^i) để xử lý mỗi khoảng trong O((5n)² log T).
💡 Tại sao ma trận 5n?
Mỗi cung tốn từ 1 đến 5 đơn vị thời gian. Để tính dp[t+1], cần biết dp[t], dp[t-1], dp[t-2], dp[t-3], dp[t-4]. Vậy vector trạng thái V_t = [dp[t], dp[t-1], dp[t-2], dp[t-3], dp[t-4]] có 5n phần tử.

Code C++

#include<bits/stdc++.h>
#define ll long long
using namespace std;
const int N = 52;
const int S = 250;
const int LOG = 30;
const ll NEG = -(1LL << 60);
int n, m, k, u[N * 12], v[N * 12], w[N * 12], cnt;
int sz;
ll T, c[N];
ll pw[LOG][S][S], tmp[S][S], dp[S], ndp[S];
struct tt { ll t, y; int x; };
tt e[205];
bool cmp(tt a, tt b){ return a.t < b.t; }

void mul(int z){
    for(int i = 0; i < sz; i++)
        for(int j = 0; j < sz; j++) tmp[i][j] = NEG;
    for(int i = 0; i < sz; i++)
        for(int j = 0; j < sz; j++){
            if(pw[z][i][j] == NEG) continue;
            ll x = pw[z][i][j];
            for(int h = 0; h < sz; h++){
                if(pw[z][j][h] == NEG) continue;
                tmp[i][h] = max(tmp[i][h], x + pw[z][j][h]);
            }
        }
    for(int i = 0; i < sz; i++)
        for(int j = 0; j < sz; j++) pw[z + 1][i][j] = tmp[i][j];
}

void apply(int z){
    for(int i = 0; i < sz; i++){
        ndp[i] = NEG;
        for(int j = 0; j < sz; j++){
            if(pw[z][i][j] == NEG || dp[j] == NEG) continue;
            ndp[i] = max(ndp[i], pw[z][i][j] + dp[j]);
        }
    }
    for(int i = 0; i < sz; i++) dp[i] = ndp[i];
}

void go(ll d){
    int bit = 0;
    while(d > 0){
        if(d & 1) apply(bit);
        d >>= 1;
        bit++;
    }
}

void step(int x, ll y){
    for(int i = 0; i < sz; i++) ndp[i] = NEG;
    for(int i = 1; i <= cnt; i++){
        int a = u[i], b = v[i], t = w[i];
        ll val = dp[(t - 1) * n + a];
        if(val == NEG) continue;
        val += c[b];
        if(b == x) val += y;
        ndp[b] = max(ndp[b], val);
    }
    for(int j = 1; j < 5; j++)
        for(int i = 0; i < n; i++)
            ndp[j * n + i] = dp[(j - 1) * n + i];
    for(int i = 0; i < sz; i++) dp[i] = ndp[i];
}

void init(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
    cin >> n >> m >> T >> k;
    for(int i = 0; i < n; i++) cin >> c[i];
    cnt = m;
    for(int i = 1; i <= m; i++){
        cin >> u[i] >> v[i] >> w[i];
        u[i]--; v[i]--;
    }
    for(int i = 1; i <= k; i++){
        cin >> e[i].t >> e[i].x >> e[i].y;
        e[i].x--;
    }
    sort(e + 1, e + k + 1, cmp);
}

void solve(){
    sz = 5 * n;
    for(int i = 0; i < sz; i++)
        for(int j = 0; j < sz; j++) pw[0][i][j] = NEG;
    for(int i = 1; i <= m; i++){
        int a = u[i], b = v[i], t = w[i];
        pw[0][b][(t - 1) * n + a] = max(pw[0][b][(t - 1) * n + a], c[b]);
    }
    for(int j = 1; j < 5; j++)
        for(int i = 0; i < n; i++)
            pw[0][j * n + i][(j - 1) * n + i] = 0;
    for(int i = 1; i < LOG; i++) mul(i - 1);
    for(int i = 0; i < sz; i++) dp[i] = NEG;
    dp[0] = c[0];
    ll cur = 0;
    for(int i = 1; i <= k; i++){
        go(e[i].t - cur - 1);
        step(e[i].x, e[i].y);
        cur = e[i].t;
    }
    go(T - cur);
    if(dp[0] == NEG) cout << -1;
    else cout << dp[0];
}

int main(){ init(); solve(); return 0; }
